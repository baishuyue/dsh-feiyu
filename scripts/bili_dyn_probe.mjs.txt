// bili_dyn_probe —— 动态发布排查：①页内 same-origin fetch ②官方发布框 UI 自动化
// 跑法：node scripts/bili_dyn_probe.mjs "<动态文案>" [--ui]
import fs from 'node:fs'
import path from 'node:path'
import os from 'node:os'

const text = (process.argv[2] || '').trim() || '【潜水日志】测试'
const useUI = process.argv.includes('--ui')

// 临时 storage state：把 B站凭据转成 playwright storageState
const cred = JSON.parse(fs.readFileSync(path.join(os.homedir(), '.feiyu-ai', 'credentials.json'), 'utf8'))
const cookies = [
  { name: 'SESSDATA', value: cred.sessdata, domain: '.bilibili.com', path: '/' },
  { name: 'bili_jct', value: cred.bili_jct, domain: '.bilibili.com', path: '/' },
  { name: 'buvid3', value: cred.buvid3, domain: '.bilibili.com', path: '/' },
  { name: 'DedeUserID', value: '3690984711915865', domain: '.bilibili.com', path: '/' },
]
const stateFile = path.join(os.tmpdir(), 'feiyu-bili-state.json')
fs.writeFileSync(stateFile, JSON.stringify({ cookies }))

const { chromium } = await import('playwright').catch(() => import('patchright'))
const browser = await chromium.launch({ headless: true, args: ['--disable-blink-features=AutomationControlled', '--no-sandbox'] })
const context = await browser.newContext({ storageState: stateFile, userAgent: 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0.0.0 Safari/537.36' })
const page = await context.new_page ?? context.newPage
await page.goto('https://t.bilibili.com/', { wait_until: 'domcontentloaded', timeout: 60000 })
await page.waitForTimeout(5000)

if (!useUI) {
  // ① 页内 fetch：same-origin，浏览器自动带全部 cookie
  const res = await page.evaluate(async (t) => {
    const dyn = { content: { contents: [{ raw_text: t, type: 1, biz_id: '' }] }, scene: 2, attach_card: {}, upload_id: Date.now(), meta: { app_meta: { from_app_show_type: '3000' } }, pics: [] }
    const body = new URLSearchParams({ dyn_req: JSON.stringify(dyn), csrf: document.cookie.match(/bili_jct=([^;]+)/)?.[1] ?? '', csrf_token: document.cookie.match(/bili_jct=([^;]+)/)?.[1] ?? '' })
    const r = await fetch('/x/dynamic/feed/create/dyn?platform=web&web_location=333.1387', { method: 'POST', credentials: 'include', headers: { 'content-type': 'application/x-www-form-urlencoded' }, body: body.toString() })
    return await r.json().catch(() => ({ code: -99 }))
  }, text)
  console.log('页内 fetch:', JSON.stringify(res).slice(0, 200))
  if (Number(res.code) === 0) { console.log('✅ 页内 fetch 成功！动态 id=' + (res.data?.dyn_id_str ?? res.data?.dyn_id ?? '?')); await browser.close(); process.exit(0) }
  console.log('页内 fetch 也 -400，切 UI 自动化…')
}

// ② 官方发布框：点「发布动态」→ 输入 → 发布
try {
  const trigger = page.locator('div.bili-dyn-publishing__trigger:visible, .bili-dyn-publishing__trigger').first
  await trigger.click({ timeout: 8000 })
  await page.waitForTimeout(2000)
  const editor = page.locator('.bili-rich-textarea:not(.disabled), [contenteditable="true"].bili-rich-textarea, .dyn-input textarea').first
  await editor.click({ timeout: 8000 })
  await page.keyboard.type(text, { delay: 30 })
  await page.waitForTimeout(800)
  await page.screenshot({ path: 'D:/social-auto-upload/_dyn_editor.png' })
  const publish = page.locator('button.publish-btn:visible, .bili-button.publish-btn:not(.disabled)').first
  await publish.click({ timeout: 8000 })
  await page.waitForTimeout(4000)
  await page.screenshot({ path: 'D:/social-auto-upload/_dyn_after.png' })
  console.log('UI 发布已点击（看截图确认）')
} catch (e) {
  console.log('UI 自动化失败：' + String(e).slice(0, 150))
  await page.screenshot({ path: 'D:/social-auto-upload/_dyn_fail.png' }).catch(() => {})
}
await browser.close()
